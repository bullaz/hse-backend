import React, { useEffect, useState } from 'react';
import { View, Text, StyleSheet } from 'react-native';
import NetInfo from '@react-native-community/netinfo';

const NetworkStatusComponent = () => {
  const [isConnected, setIsConnected] = useState(true);

  useEffect(() => {
    const unsubscribe = NetInfo.addEventListener((state) => {
      setIsConnected(state.isConnected);
      // state.isInternetReachable provides a more reliable check for actual internet access
    });

    // Unsubscribe from events when component unmounts
    return () => {
      unsubscribe();
    };
  }, []);

  return (
    <View style={styles.container}>
      <Text style={styles.statusText}>
        {isConnected ? '✅ Online' : '❌ Offline'}
      </Text>
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    flex: 1,
    justifyContent: 'center',
    alignItems: 'center',
  },
  statusText: {
    fontSize: 20,
    fontWeight: 'bold',
  },
});

export default NetworkStatusComponent;

I just took off the value prop and it works
Good solution! Removing the value prop is actually a valid workaround for this library. Here's why it works:

Why Removing value Works:
The dropdown manages its own internal state when you don't explicitly control it with a value prop

No more object reference comparison issues - the library handles selection internally

Your onChange handler still works to update your formData state

Potential Downsides to Consider:
Two-way binding is lost - if you programmatically change formData.societe, the dropdown won't reflect it

No initial/default value - you can't preset the dropdown selection

If you reset the form, the dropdown won't clear automatically
